def preprocess_frame(frame):

    # OpenCV BGR -> RGB
    frame = cv2.cvtColor(
        frame,
        cv2.COLOR_BGR2RGB
    )

    frame = cv2.resize(
        frame,
        (
            IMG_SIZE,
            IMG_SIZE
        ),
        interpolation=cv2.INTER_AREA
    )

    frame = frame.astype(
        np.float32
    )

    frame /= 255.0

    return frame


# ============================================================
# EXTRACT VIDEO SEQUENCE
# ============================================================

def extract_video_sequence(
    video_path
):

    cap = cv2.VideoCapture(
        video_path
    )

    if not cap.isOpened():

        raise ValueError(
            "Unable to open video."
        )

    total_frames = int(
        cap.get(
            cv2.CAP_PROP_FRAME_COUNT
        )
    )

    if total_frames <= 0:

        cap.release()

        raise ValueError(
            "Video has no readable frames."
        )

    # Select 20 evenly spaced frames
    frame_indices = np.linspace(
        0,
        total_frames - 1,
        SEQUENCE_LENGTH
    ).astype(int)

    frames = []

    for index in frame_indices:

        cap.set(
            cv2.CAP_PROP_POS_FRAMES,
            int(index)
        )

        ret, frame = cap.read()

        if not ret:

            continue

        processed = preprocess_frame(
            frame
        )

        frames.append(
            processed
        )

    cap.release()

    if len(frames) == 0:

        raise ValueError(
            "Could not extract frames."
        )

    # Pad missing frames
    while len(frames) < SEQUENCE_LENGTH:

        frames.append(
            frames[-1].copy()
        )

    frames = frames[
        :SEQUENCE_LENGTH
    ]

    return np.asarray(
        frames,
        dtype=np.float32
    )


# ============================================================
# PREDICT
# ============================================================

def predict_sequence(
    sequence
):

    input_data = np.expand_dims(
        sequence,
        axis=0
    )

    prediction = model.predict(
        input_data,
        verbose=0
    )

    # Model has sigmoid output
    fake_probability = float(
        np.asarray(
            prediction
        ).reshape(-1)[0]
    )

    fake_probability = max(
        0.0,
        min(
            1.0,
            fake_probability
        )
    )

    real_probability = (
        1.0 -
        fake_probability
    )

    is_fake = (
        fake_probability >= 0.5
    )

    confidence = (
        fake_probability
        if is_fake
        else real_probability
    )

    return {

        "is_fake":
            bool(is_fake),

        "label":
            "FAKE"
            if is_fake
            else "REAL",

        "confidence":
            round(
                confidence,
                4
            ),

        "fake_probability":
            round(
                fake_probability,
                4
            ),

        "real_probability":
            round(
                real_probability,
                4
            )
    }


# ============================================================

def predict_image(
    image_path
):

    image = cv2.imread(
        image_path
    )

    if image is None:

        raise ValueError(
            "Unable to read image."
        )

    frame = preprocess_frame(
        image
    )

    # Repeat image 20 times
    sequence = np.repeat(
        frame[np.newaxis, ...],
        SEQUENCE_LENGTH,
        axis=0
    )

    return predict_sequence(
        sequence
    )


# ============================================================
# VIDEO PREDICTION
# ============================================================

def predict_video(
    video_path
):

    sequence = extract_video_sequence(
        video_path
    )

    return predict_sequence(
        sequence
    )

@app.route(
    "/predict",
    methods=["POST"]
)
def predict():

    # --------------------------------------------------------
    # Accept:
    #
    # file
    # image
    # video
    # --------------------------------------------------------

    uploaded_file = None

    if "file" in request.files:

        uploaded_file = request.files[
            "file"
        ]

    elif "image" in request.files:

        uploaded_file = request.files[
            "image"
        ]

    elif "video" in request.files:

        uploaded_file = request.files[
            "video"
        ]

    if uploaded_file is None:

        return jsonify({

            "success": False,

            "error":
                "No file uploaded."
        }), 400

    if uploaded_file.filename == "":

        return jsonify({

            "success": False,

            "error":
                "No file selected."
        }), 400


    filename = uploaded_file.filename

    extension = os.path.splitext(
        filename
    )[1].lower()


    image_extensions = {
        ".jpg",
        ".jpeg",
        ".png",
        ".bmp",
        ".webp"
    }

    video_extensions = {
        ".mp4",
        ".avi",
        ".mov",
        ".mkv",
        ".webm"
    }


    if extension in image_extensions:

        file_type = "image"

    elif extension in video_extensions:

        file_type = "video"

    else:

        return jsonify({

            "success": False,

            "error":
                "Unsupported file type."
        }), 400


    temp_path = None


    try:

        # ----------------------------------------------------
        # Temporary file
        # ----------------------------------------------------

        with tempfile.NamedTemporaryFile(
            delete=False,
            suffix=extension
        ) as temp:

            uploaded_file.save(
                temp.name
            )

            temp_path = temp.name


        # ----------------------------------------------------
        # Prediction
        # ----------------------------------------------------

        start_time = time.time()

        if file_type == "image":

            result = predict_image(
                temp_path
            )

        else:

            result = predict_video(
                temp_path
            )


        processing_time = (
            time.time()
            -
            start_time
        )


        # ----------------------------------------------------
        # Response
        # ----------------------------------------------------

        result.update({

            "success": True,

            "file_name":
                filename,

            "file_type":
                file_type,

            "processing_time":
                round(
                    processing_time,
                    3
                )
        })


        return jsonify(
            result
        ), 200


    except Exception as e:

        print(
            "\nPrediction error:"
        )

        print(
            str(e)
        )

        traceback.print_exc()

        return jsonify({

            "success": False,

            "error":
                str(e)
        }), 500


    finally:

        if (
            temp_path
            and os.path.exists(
                temp_path
            )
        ):

            try:

                os.remove(
                    temp_path
                )

            except Exception:

                pass

import os
import numpy as np
import tensorflow as tf


class VideoSequenceGenerator(tf.keras.utils.Sequence):

    def __init__(
        self,
        real_dir,
        fake_dir,
        batch_size=1,
        shuffle=True
    ):

        self.real_dir = real_dir
        self.fake_dir = fake_dir
        self.batch_size = batch_size
        self.shuffle = shuffle

        self.file_paths = []
        self.labels = []

        # ============================================
        # LOAD REAL DATA
        # ============================================

        if os.path.exists(real_dir):

            for filename in os.listdir(real_dir):

                if filename.lower().endswith(".npy"):

                    path = os.path.join(
                        real_dir,
                        filename
                    )

                    self.file_paths.append(path)
                    self.labels.append(0)

        # ============================================
        # LOAD FAKE DATA
        # ============================================

        if os.path.exists(fake_dir):

            for filename in os.listdir(fake_dir):

                if filename.lower().endswith(".npy"):

                    path = os.path.join(
                        fake_dir,
                        filename
                    )

                    self.file_paths.append(path)
                    self.labels.append(1)

        self.file_paths = np.array(
            self.file_paths
        )

        self.labels = np.array(
            self.labels,
            dtype=np.float32
        )

        self.indices = np.arange(
            len(self.file_paths)
        )

        self.on_epoch_end()

        print(
            f"Generator loaded: "
            f"{len(self.file_paths)} samples"
        )

        print(
            f"Real: {np.sum(self.labels == 0)}"
        )

        print(
            f"Fake: {np.sum(self.labels == 1)}"
        )


    def __len__(self):

        return int(
            np.ceil(
                len(self.file_paths)
                / self.batch_size
            )
        )


    def __getitem__(self, index):

        start = (
            index *
            self.batch_size
        )

        end = min(
            start +
            self.batch_size,
            len(self.file_paths)
        )

        batch_indices = self.indices[
            start:end
        ]

        batch_paths = self.file_paths[
            batch_indices
        ]

        batch_labels = self.labels[
            batch_indices
        ]

        batch_x = []

        for path in batch_paths:

            try:

                sequence = np.load(
                    path
                ).astype(
                    np.float32
                )

                # ====================================
                # EXPECTED SHAPE:
                #
                # (20, 128, 128, 3)
                # ====================================

                if sequence.ndim != 4:

                    raise ValueError(
                        f"Invalid shape {sequence.shape}"
                    )

                # Normalize if stored as uint8
                if sequence.max() > 1.0:

                    sequence = (
                        sequence / 255.0
                    )

                batch_x.append(
                    sequence
                )

            except Exception as e:

                print(
                    f"Error loading {path}: {e}"
                )

                # Fallback
                batch_x.append(
                    np.zeros(
                        (20, 128, 128, 3),
                        dtype=np.float32
                    )
                )

        batch_x = np.asarray(
            batch_x,
            dtype=np.float32
        )

        batch_labels = np.asarray(
            batch_labels,
            dtype=np.float32
        )

        return batch_x, batch_labels


    def on_epoch_end(self):

        if self.shuffle:

            np.random.shuffle(
                self.indices
            )

import cv2
import numpy as np
import os
import gc

# CONFIGURATION
# ============================================================

SEQUENCE_LENGTH = 20

IMG_SIZE = 128

MAX_FRAMES = 30

FRAME_SKIP = 6

MAX_DIMENSION = 320


# ============================================================
# FACE EXTRACTION
# ============================================================

def extract_faces_haar(video_path):

    cap = cv2.VideoCapture(
        video_path
    )

    if not cap.isOpened():

        return None

    face_cascade = cv2.CascadeClassifier(
        cv2.data.haarcascades +
        "haarcascade_frontalface_default.xml"
    )

    faces = []

    frame_count = 0

    while (
        len(faces) < SEQUENCE_LENGTH
        and frame_count < MAX_FRAMES
    ):

        ret, frame = cap.read()

        if not ret:

            break

        frame_count += 1

        # Skip frames
        if frame_count % FRAME_SKIP != 0:

            continue

        # ============================================
        # RESIZE LARGE FRAME
        # ============================================

        try:

            h, w = frame.shape[:2]

            if max(h, w) > MAX_DIMENSION:

                scale = (
                    MAX_DIMENSION /
                    max(h, w)
                )

                new_w = int(
                    w * scale
                )

                new_h = int(
                    h * scale
                )

                frame = cv2.resize(
                    frame,
                    (new_w, new_h),
                    interpolation=cv2.INTER_AREA
                )

        except Exception:

            continue

        # ============================================
        # FACE DETECTION
        # ============================================

        try:

            gray = cv2.cvtColor(
                frame,
                cv2.COLOR_BGR2GRAY
            )

            detected = face_cascade.detectMultiScale(
                gray,
                scaleFactor=1.1,
                minNeighbors=4,
                minSize=(30, 30)
            )

        except Exception:

            continue

        if len(detected) == 0:

            continue

        # ============================================
        # LARGEST FACE
        # ============================================

        x, y, w, h = max(
            detected,
            key=lambda rect:
                rect[2] * rect[3]
        )

        face = frame[
            y:y+h,
            x:x+w
        ]

        if face.size == 0:

            continue

        try:

            face = cv2.resize(
                face,
                (IMG_SIZE, IMG_SIZE),
                interpolation=cv2.INTER_AREA
            )

            faces.append(
                face
            )

        except Exception:

            continue

    cap.release()

    gc.collect()

    # ================================================
    # PAD IF FEWER THAN 20 FACES
    # ================================================

    if len(faces) > 0:

        while len(faces) < SEQUENCE_LENGTH:

            faces.append(
                faces[-1].copy()
            )

    if len(faces) == 0:

        return None

    return np.asarray(
        faces,
        dtype=np.uint8
    )

def process_directory(
    input_dir,
    output_dir,
    label_name
):

    if not os.path.exists(input_dir):

        print(
            f"ERROR: Directory not found:"
            f"\n{input_dir}"
        )

        return 0, 0

    saved = 0
    skipped = 0

    videos = sorted(
        os.listdir(input_dir)
    )

    for index, video in enumerate(videos):

        if not video.lower().endswith(
            (
                ".mp4",
                ".avi",
                ".mov",
                ".mkv",
                ".webm"
            )
        ):

            continue

        video_path = os.path.join(
            input_dir,
            video
        )

        print(
            f"[{index + 1}] "
            f"Processing {label_name}: "
            f"{video}"
        )

        try:

            sequence = extract_faces_haar(
                video_path
            )

            if (
                sequence is None
                or len(sequence) != SEQUENCE_LENGTH
            ):

                print(
                    "   SKIPPED - insufficient faces"
                )

                skipped += 1

                continue

            # Use original filename
            base_name = os.path.splitext(
                video
            )[0]

            output_path = os.path.join(
                output_dir,
                base_name + ".npy"
            )

            np.save(
                output_path,
                sequence
            )

            print(
                f"   SAVED -> {output_path}"
            )

            saved += 1

        except Exception as e:

            print(
                f"   ERROR: {e}"
            )

            skipped += 1

    return saved, skipped

def build_model():

    inputs = tf.keras.Input(
        shape=(
            SEQUENCE_LENGTH,
            IMG_SIZE,
            IMG_SIZE,
            CHANNELS
        )
    )

    # ================================================
    # CNN FEATURE EXTRACTION
    # ================================================

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.Conv2D(
            32,
            (3, 3),
            activation="relu",
            padding="same"
        )
    )(inputs)

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.MaxPooling2D(
            (2, 2)
        )
    )(x)

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.Conv2D(
            64,
            (3, 3),
            activation="relu",
            padding="same"
        )
    )(x)

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.MaxPooling2D(
            (2, 2)
        )
    )(x)

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.Conv2D(
            128,
            (3, 3),
            activation="relu",
            padding="same"
        )
    )(x)

    x = tf.keras.layers.TimeDistributed(
        tf.keras.layers.GlobalAveragePooling2D()
    )(x)

    # ================================================
    # TEMPORAL MODEL
    # ================================================

    x = tf.keras.layers.LSTM(
        128,
        return_sequences=False
    )(x)

    x = tf.keras.layers.Dropout(
        0.5
    )(x)

    x = tf.keras.layers.Dense(
        64,
        activation="relu"
    )(x)

    x = tf.keras.layers.Dropout(
        0.3
    )(x)

    outputs = tf.keras.layers.Dense(
        1,
        activation="sigmoid"
    )(x)

    model = tf.keras.Model(
        inputs,
        outputs
    )

    model.compile(
        optimizer=tf.keras.optimizers.Adam(
            learning_rate=LEARNING_RATE
        ),
        loss="binary_crossentropy",
        metrics=[
            "accuracy"
        ]
    )

    return model

        test_labels
    ) = train_test_split(
        paths,
        labels,
        test_size=0.2,
        random_state=RANDOM_STATE,
        stratify=labels
    )

    (
        train_paths,
        val_paths,
        train_labels,
        val_labels
    ) = train_test_split(
        train_paths,
        train_labels,
        test_size=0.2,
        random_state=RANDOM_STATE,
        stratify=train_labels
    )

    print(
        f"\nTraining samples: "
        f"{len(train_paths)}"
    )

    print(
        f"Validation samples: "
        f"{len(val_paths)}"
    )

    print(
        f"Test samples: "
        f"{len(test_paths)}"
    )

    # --------------------------------------------
    # Load arrays
    # --------------------------------------------

    print("\nLoading training data...")

    X_train = load_sequences(
        train_paths
    )

    y_train = train_labels[
        :len(X_train)
    ]

    print("\nLoading validation data...")

    X_val = load_sequences(
        val_paths
    )

    y_val = val_labels[
        :len(X_val)
    ]

    # --------------------------------------------
    # Model
    # --------------------------------------------

    print("\nBuilding model...")

    model = build_model()

    model.summary()

    # --------------------------------------------
    # Callbacks
    # --------------------------------------------

    callbacks = [

        tf.keras.callbacks.ModelCheckpoint(
            MODEL_PATH,
            monitor="val_accuracy",
            save_best_only=True,
            mode="max",
            verbose=1
        ),

        tf.keras.callbacks.EarlyStopping(
            monitor="val_loss",
            patience=3,
            restore_best_weights=True,
            verbose=1
        ),

        tf.keras.callbacks.ReduceLROnPlateau(
            monitor="val_loss",
            factor=0.5,
            patience=2,
            min_lr=1e-7,
            verbose=1
        )
    ]

    # --------------------------------------------
    # Train
    # --------------------------------------------

    print("\nStarting training...")

    history = model.fit(

        X_train,
        y_train,

        validation_data=(
            X_val,
            y_val
        ),

        epochs=EPOCHS,

        batch_size=BATCH_SIZE,

        callbacks=callbacks,

        verbose=1
    )

    # --------------------------------------------
    # Load best model
    # --------------------------------------------

    if os.path.exists(
        MODEL_PATH
    ):

        model = tf.keras.models.load_model(
            MODEL_PATH
        )

    ]

import React, { useRef, useState, useCallback } from 'react';
import { Button, Alert, Spinner } from 'react-bootstrap';

interface CameraCaptureProps {
  onCapture: (videoBlob: Blob) => void;
  isAnalyzing?: boolean;
}

const CameraCapture: React.FC<CameraCaptureProps> = ({ onCapture, isAnalyzing = false }) => {
  const videoRef = useRef<HTMLVideoElement>(null);
  const mediaStreamRef = useRef<MediaStream | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [hasPermission, setHasPermission] = useState(false);

  const startCamera = useCallback(async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ video: true });
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        mediaStreamRef.current = stream;
        setHasPermission(true);
        setError(null);
      }
    } catch (err) {
      setError('Unable to access camera. Please check permissions.');
      console.error(err);
    }
  }, []);

  const stopCamera = useCallback(() => {
    if (mediaStreamRef.current) {
      mediaStreamRef.current.getTracks().forEach(track => track.stop());
      mediaStreamRef.current = null;
    }
    if (videoRef.current) {
      videoRef.current.srcObject = null;
    }
    setHasPermission(false);
  }, []);

  const captureVideo = useCallback(() => {
    if (!videoRef.current || !videoRef.current.srcObject) return;

    const stream = videoRef.current.srcObject as MediaStream;
    const mediaRecorder = new MediaRecorder(stream, { mimeType: 'video/webm' });
    const chunks: BlobPart[] = [];

    mediaRecorder.ondataavailable = (e) => chunks.push(e.data);
    mediaRecorder.onstop = () => {
      const blob = new Blob(chunks, { type: 'video/webm' });
      onCapture(blob);
    };

    mediaRecorder.start();
    setTimeout(() => {
      mediaRecorder.stop();
    }, 3000); // record 3 seconds
  }, [onCapture]);

  return (
    <div className="text-center">
      {!hasPermission ? (
        <Button variant="primary" onClick={startCamera} className="mb-3">
          📷 Start Camera
        </Button>
      ) : (
        <div>
          <video
            ref={videoRef}
            autoPlay
            playsInline
            muted
            style={{ width: '100%', maxHeight: '400px', borderRadius: '12px' }}
          />
          <div className="mt-3">
            <Button variant="success" onClick={captureVideo} disabled={isAnalyzing} className="me-2">
              🎥 Capture 3s Video
            </Button>
            <Button variant="secondary" onClick={stopCamera}>
              Stop Camera
            </Button>
          </div>
        </div>
      )}
      {isAnalyzing && <Spinner animation="border" variant="primary" className="mt-3" />}
      {error && <Alert variant="danger" className="mt-3">{error}</Alert>}
    </div>
  );
};

export default CameraCapture;

import React, { useState, useRef, useEffect } from 'react';
import { Button, Card, Form, InputGroup, Spinner } from 'react-bootstrap';
import axios from 'axios';

interface Message {
  text: string;
  sender: 'user' | 'bot';
  source?: string;
}

const Chatbot: React.FC = () => {
  const [isOpen, setIsOpen] = useState(false);
  const [messages, setMessages] = useState<Message[]>([
    { text: "Hi! Ask me anything about deepfakes, detection, or ethics.", sender: 'bot' }
  ]);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const chatEndRef = useRef<HTMLDivElement>(null);

  const scrollToBottom = () => {
    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    scrollToBottom();
  }, [messages]);

  const handleSend = async () => {
    if (!input.trim() || loading) return;
    const userMsg: Message = { text: input, sender: 'user' };
    setMessages(prev => [...prev, userMsg]);
    setInput('');
    setLoading(true);

    try {
      const response = await axios.post('http://localhost:5000/chat', { query: input });
      const botReply = response.data.response || "I'm not sure. Please ask something else.";
      const source = response.data.source || '';
      setMessages(prev => [...prev, { text: botReply, sender: 'bot', source }]);
    } catch (err) {
      setMessages(prev => [...prev, { text: "Sorry, I'm having trouble right now. Please try again.", sender: 'bot' }]);
    }
    setLoading(false);
  };

  return (
    <>
      <Button
        variant="primary"
        className="rounded-circle position-fixed"
        style={{ bottom: '20px', right: '20px', zIndex: 999, width: '60px', height: '60px' }}
        onClick={() => setIsOpen(!isOpen)}
      >
        {isOpen ? <i className="bi bi-x-lg"></i> : <i className="bi bi-chat-dots"></i>}
      </Button>

      {isOpen && (
        <Card
          className="shadow-lg position-fixed"
          style={{ bottom: '100px', right: '20px', zIndex: 999, width: '380px', maxHeight: '550px' }}
        >
          <Card.Header className="bg-primary text-white d-flex justify-content-between align-items-center">
            <span><i className="bi bi-robot me-2"></i>Deepfake Assistant</span>
            <small className="text-light">Powered by AI</small>
          </Card.Header>
          <Card.Body style={{ overflowY: 'auto', maxHeight: '380px' }}>
            {messages.map((msg, idx) => (
              <div key={idx} className={`d-flex ${msg.sender === 'user' ? 'justify-content-end' : 'justify-content-start'} mb-2`}>
                <div
                  className={`p-2 rounded-3 ${msg.sender === 'user' ? 'bg-primary text-white' : 'bg-light border'}`}
                  style={{ maxWidth: '80%' }}
                >
                  {msg.text}
                  {msg.source && msg.source === 'ai' && (
                    <div className="mt-1"><small className="text-muted fst-italic">AI‑generated</small></div>
                  )}
                </div>
              </div>
            ))}
            {loading && (
              <div className="d-flex justify-content-start mb-2">
                <div className="p-2 bg-light border rounded-3">
                  <Spinner animation="grow" size="sm" />
                  <Spinner animation="grow" size="sm" className="ms-1" />
                  <Spinner animation="grow" size="sm" className="ms-1" />
                </div>
              </div>
            )}
            <div ref={chatEndRef} />
          </Card.Body>
          <Card.Footer>
            <InputGroup>
              <Form.Control
                type="text"
                placeholder="Ask about deepfakes..."
                value={input}
                onChange={(e) => setInput(e.target.value)}
                onKeyPress={(e) => e.key === 'Enter' && handleSend()}
                disabled={loading}
              />
              <Button variant="primary" onClick={handleSend} disabled={loading}>
                {loading ? <Spinner animation="border" size="sm" /> : <i className="bi bi-send"></i>}
              </Button>
            </InputGroup>
          </Card.Footer>
        </Card>
      )}
    </>
  );
};

export default Chatbot;

import React, { useState, useRef } from 'react';
import axios from 'axios';
import { Container, Row, Col, Card, Form, Button, Alert, Spinner } from 'react-bootstrap';
import './AddVideo.css'
const AddVideo: React.FC = () => {
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [uploading, setUploading] = useState(false);
  const [message, setMessage] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [detectionResult, setDetectionResult] = useState<{ is_fake: boolean; confidence: number } | null>(null);
  const [needsConfirmation, setNeedsConfirmation] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleFileChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (file && file.type.startsWith('video/')) {
      setSelectedFile(file);
      setMessage(null);
      setError(null);
      setDetectionResult(null);
      setNeedsConfirmation(false);
    } else {
      setError('Please upload a valid video file (MP4, AVI, MOV)');
      event.target.value = '';
    }
  };

  const handleAnalyze = async () => {
    if (!selectedFile) {
      setError('Please select a video');
      return;
    }
    const formData = new FormData();
    formData.append('video', selectedFile);
    formData.append('confirmed', 'false');

    setUploading(true);
    setMessage(null);
    setError(null);
    setNeedsConfirmation(false);

    try {
      const response = await axios.post('http://localhost:5000/add_video_auto', formData);
      if (response.data.needs_confirmation) {
        setNeedsConfirmation(true);
        setDetectionResult({
          is_fake: response.data.is_fake,
          confidence: response.data.confidence
        });
      } else {
        setMessage(response.data.message);
      }
    } catch (err: any) {
      setError(err.response?.data?.error || 'Analysis failed');
    } finally {
      setUploading(false);
    }
  };

  const handleConfirm = async () => {
    if (!selectedFile) return;
    const formData = new FormData();
    formData.append('video', selectedFile);
    formData.append('confirmed', 'true');

    setUploading(true);
    setMessage(null);
    setError(null);

    try {
      const response = await axios.post('http://localhost:5000/add_video_auto', formData);
      setMessage(response.data.message);
      setSelectedFile(null);
      setDetectionResult(null);
      setNeedsConfirmation(false);
      if (fileInputRef.current) fileInputRef.current.value = '';
    } catch (err: any) {
      setError(err.response?.data?.error || 'Upload failed');
    } finally {
      setUploading(false);
    }
  };

  const handleDropZoneClick = () => {
    fileInputRef.current?.click();
  };

  return (
    <Container className="py-5">
      <Row className="justify-content-center">
        <Col lg={8}>
          <Card className="shadow-lg border-0 rounded-4">
            <div className="bg-gradient-primary p-4 text-white text-center">
              <h1 className="display-5 fw-bold">📤 Add Video to Dataset</h1>
              <p className="lead mb-0">Upload a video – our AI will detect if it's real or fake, then you confirm.</p>
            </div>
            <Card.Body className="p-4">
              <Form>
                <div
                  className="dropzone p-5 text-center rounded-4 border-2 border-dashed border-secondary"
                  style={{ cursor: 'pointer', transition: 'all 0.2s' }}
                  onClick={handleDropZoneClick}
                  onDragOver={(e) => e.preventDefault()}
                  onDrop={(e) => {
                    e.preventDefault();
                    const file = e.dataTransfer.files[0];
                    if (file && file.type.startsWith('video/')) {
                      setSelectedFile(file);
                      setMessage(null);
                      setError(null);
                      setDetectionResult(null);
                      setNeedsConfirmation(false);
                    } else {
                      setError('Please upload a valid video file (MP4, AVI, MOV)');
                    }
                  }}
                >
                  <input
                    type="file"
                    ref={fileInputRef}
                    accept="video/*"
                    onChange={handleFileChange}
                    style={{ display: 'none' }}
                  />
                  <i className="bi bi-cloud-upload fs-1"></i>
                  <p className="mt-2 mb-0">Drag & drop a video, or click to select</p>
                  <small className="text-muted">MP4, AVI, MOV</small>
                </div>
                {selectedFile && (
                  <div className="mt-3">
                    <p><strong>Selected:</strong> {selectedFile.name}</p>
                  </div>
                )}
                
                {!needsConfirmation && !message && (
                  <Button
                    variant="primary"
                    onClick={handleAnalyze}
                    className="mt-3 w-100"
                    disabled={uploading || !selectedFile}
                  >
                    {uploading ? <Spinner animation="border" size="sm" /> : '🔍 Analyze & Detect'}
                  </Button>
                )}

                {needsConfirmation && detectionResult && (
                  <div className="mt-3 p-3 rounded-3 bg-light">
                    <Alert variant={detectionResult.is_fake ? 'danger' : 'success'}>
                      <h5>
                        {detectionResult.is_fake 
                          ? '🚨 AI detects this as FAKE' 
                          : '✅ AI detects this as REAL'}
                      </h5>
                      <p>Confidence: {(detectionResult.confidence * 100).toFixed(2)}%</p>
                      <p className="mb-0">Do you want to save this video as <strong>{detectionResult.is_fake ? 'FAKE' : 'REAL'}</strong>?</p>
                    </Alert>
                    <div className="d-flex gap-2">
                      <Button variant="success" onClick={handleConfirm} disabled={uploading} className="flex-1">
                        {uploading ? <Spinner animation="border" size="sm" /> : '✅ Confirm & Save'}
                      </Button>
                      <Button variant="secondary" onClick={() => {
                        setNeedsConfirmation(false);
                        setDetectionResult(null);
                        setSelectedFile(null);
                        if (fileInputRef.current) fileInputRef.current.value = '';
                      }} className="flex-1">
                        ❌ Cancel
                      </Button>
                    </div>
                  </div>
                )}

                {message && <Alert variant="success" className="mt-3">{message}</Alert>}

  raw_score: number;

              )}
              {error && <Alert variant="danger" className="mt-3">{error}</Alert>}
            </Card.Body>
          </Card>
        </Col>

import React, { useState, useEffect, useRef } from 'react';
import axios from 'axios';
import { Container, Row, Col, Card, Button, Alert, Spinner, ProgressBar, Form } from 'react-bootstrap';
import { CircularProgressbar, buildStyles } from 'react-circular-progressbar';
import 'react-circular-progressbar/dist/styles.css';
import './Upload.css'
interface PredictionResult {
  is_fake: boolean;
  confidence: number;
  raw_score: number;
  frame_scores?: number[];
  suspicious_frames?: number[];
  num_frames?: number;
  audio_features?: any;
}

interface ModelMetrics {
  accuracy: number;
  precision: number;
  recall: number;
  f1_score: number;
  test_samples: number;
}

const Upload: React.FC = () => {
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [fileType, setFileType] = useState<'video' | 'image' | null>(null);
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<PredictionResult | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [metrics, setMetrics] = useState<ModelMetrics | null>(null);
  const [showFrameDetails, setShowFrameDetails] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    axios.get('http://localhost:5000/metrics')
      .then(res => setMetrics(res.data))
      .catch(err => console.error('Could not load metrics:', err));
  }, []);

  const handleFileChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (file) {
      if (file.type.startsWith('video/')) {
        setFileType('video');
      } else if (file.type.startsWith('image/')) {
        setFileType('image');
      } else {
        setError('Please upload a video or image file');
        return;
      }
      setSelectedFile(file);
      setPreviewUrl(URL.createObjectURL(file));
      setResult(null);
      setError(null);
    }
  };

  const handleUpload = async () => {
    if (!selectedFile) return;
    const formData = new FormData();
    formData.append(fileType === 'image' ? 'image' : 'video', selectedFile);
    setLoading(true);
    setError(null);
    setResult(null);
    try {
      const endpoint = fileType === 'image' ? '/predict_image' : '/predict_video';
      const response = await axios.post(`http://localhost:5000${endpoint}`, formData, {
        headers: { 'Content-Type': 'multipart/form-data' }
      });
      setResult(response.data);
    } catch (err: any) {
      setError(err.response?.data?.error || 'Analysis failed');
    } finally {
      setLoading(false);
    }
  };

  const confidencePercent = result ? result.confidence * 100 : 0;
  const confidenceColor = result ? (result.is_fake ? '#dc3545' : '#28a745') : '#6c757d';

  return (
    <Container className="py-5">
      <Row className="justify-content-center">
        <Col lg={8}>
          <Card className="shadow-lg border-0 rounded-4 overflow-hidden mb-4">
            <div className="bg-gradient-primary p-4 text-white text-center">
              <h1 className="display-5 fw-bold">Upload & Analyze</h1>
              <p className="lead mb-0">Upload a video or image to check its authenticity</p>
            </div>
            <Card.Body className="p-4">
              <div
                className="dropzone"
                onClick={() => fileInputRef.current?.click()}
                onDragOver={(e) => e.preventDefault()}
                onDrop={(e) => {
                  e.preventDefault();
                  const file = e.dataTransfer.files[0];
                  if (file) {
                    if (file.type.startsWith('video/')) setFileType('video');
                    else if (file.type.startsWith('image/')) setFileType('image');
                    else { setError('Please upload a video or image file'); return; }
                    setSelectedFile(file);
                    setPreviewUrl(URL.createObjectURL(file));
                    setResult(null);
                    setError(null);
                  }
                }}
              >
                <input
                  type="file"
                  ref={fileInputRef}
                  accept="video/*,image/*"
                  onChange={handleFileChange}
                  style={{ display: 'none' }}
                />
                <i className="bi bi-cloud-upload fs-1"></i>
                <p className="mt-2 mb-0">Drag & drop a video or image, or click to select</p>
                <small className="text-muted">MP4, AVI, MOV, JPG, PNG, WEBP</small>
              </div>

              {previewUrl && (
                <div className="mt-4">
                  {fileType === 'video' ? (
                    <video className="w-100 rounded-3 shadow-sm" controls src={previewUrl} style={{ maxHeight: '400px' }} />
                  ) : (
                    <img src={previewUrl} alt="Preview" className="w-100 rounded-3 shadow-sm" style={{ maxHeight: '400px', objectFit: 'contain' }} />
                  )}
                </div>
              )}

              {selectedFile && !loading && !result && (
                <div className="text-center mt-4">
                  <Button variant="primary" size="lg" onClick={handleUpload} className="px-5 rounded-pill">
                    Analyze {fileType === 'image' ? 'Image' : 'Video'}
                  </Button>
                </div>
              )}

              {loading && (
                <div className="text-center mt-4">
                  <Spinner animation="border" variant="primary" />
                  <p className="mt-2">Analyzing... this may take a moment.</p>
                  <ProgressBar animated now={100} className="w-50 mx-auto" />
                </div>
              )}

              {error && <Alert variant="danger" className="mt-4">{error}</Alert>}

              {result && (
                <div className="mt-4 p-4 rounded-4 bg-light text-center">
                  <div className="d-flex justify-content-center mb-3">
                    <div style={{ width: 150, height: 150 }}>
                      <CircularProgressbar
                        value={confidencePercent}
                        text={`${Math.round(confidencePercent)}%`}
                        styles={buildStyles({
                          textSize: '20px',
                          pathColor: confidenceColor,
                          textColor: confidenceColor,
                          trailColor: '#e9ecef'
                        })}
                      />
                    </div>
                  </div>
                  <h2 className={result.is_fake ? 'text-danger' : 'text-success'}>
                    {result.is_fake ? '🚨 DEEPFAKE DETECTED' : '✅ MEDIA APPEARS REAL'}
                  </h2>
                  <p className="lead">Confidence: {confidencePercent.toFixed(2)}%</p>
                  <p className="text-muted">Raw score: {result.raw_score.toFixed(4)} (0=Real, 1=Fake)</p>
                  {fileType === 'video' && result.frame_scores && (
                    <Button variant="outline-secondary" onClick={() => setShowFrameDetails(!showFrameDetails)} className="mt-2">
                      {showFrameDetails ? 'Hide Frame Details' : 'Show Frame Details'}
                    </Button>
                  )}
                  <Button variant="outline-secondary" onClick={() => { setSelectedFile(null); setPreviewUrl(null); setResult(null); setFileType(null); }} className="mt-2 ms-2">
                    Test Another
                  </Button>
                </div>
              )}

              {showFrameDetails && result && result.frame_scores && (
                <div className="mt-4">
                  <h5>Frame‑by‑Frame Analysis</h5>
                  <p className="text-muted small">Green = Real, Red = Fake. The model analyzes each frame independently.</p>
                  <div className="d-flex flex-wrap" style={{ maxHeight: '200px', overflowY: 'auto' }}>
                    {result.frame_scores.map((score, idx) => (
                      <div
                        key={idx}
                        className="m-1 p-1 text-center"
                        style={{
                          width: '30px',
                          height: '30px',
                          borderRadius: '4px',
                          background: `rgba(255, 0, 0, ${score})`
                        }}
                        title={`Frame ${idx + 1}: ${Math.round(score * 100)}% fake`}
                      >
                        <small style={{ color: score > 0.5 ? 'white' : 'black', fontSize: '8px' }}>
                          {idx + 1}
                        </small>
                      </div>
                    ))}
                  </div>
                  {result.suspicious_frames && result.suspicious_frames.length > 0 && (
                    <div className="mt-3">
                      <Alert variant="warning">
                        <i className="bi bi-exclamation-triangle me-2"></i>
                        <strong>Suspicious frames:</strong> {result.suspicious_frames.map(f => `Frame ${f + 1}`).join(', ')}
                        <br />
                        <small>These frames show high manipulation probability (70%).</small>
                      </Alert>
                    </div>
                  )}
                  {result.audio_features && result.audio_features.has_audio && (
                    <div className="mt-3">
                      <Alert variant={result.audio_features.is_suspicious ? 'warning' : 'info'}>
                        <i className="bi bi-music-note me-2"></i>
                        <strong>Audio Analysis:</strong>{' '}
                        {result.audio_features.is_suspicious ? 'Unusual audio patterns detected.' : 'Audio appears normal.'}
                        <br />
                        <small>RMS: {result.audio_features.rms?.toFixed(4)} | Spectral Centroid: {result.audio_features.spectral_centroid?.toFixed(0)}</small>
                      </Alert>
                    </div>
                  )}
                </div>
              )}
            </Card.Body>
          </Card>

          {metrics && (
            <Card className="shadow-sm border-0 rounded-4">
              <Card.Body>
                <h4 className="mb-3">Model Performance (on test set)</h4>
                <Row>
                  <Col md={3} sm={6} className="mb-3">
                    <div className="metric-card text-center">
                      <h6 className="text-muted">Accuracy</h6>
                      <h3 className="text-primary">{(metrics.accuracy * 100).toFixed(1)}%</h3>
                    </div>
                  </Col>
                  <Col md={3} sm={6} className="mb-3">
                    <div className="metric-card text-center">
                      <h6 className="text-muted">Precision</h6>
                      <h3 className="text-success">{(metrics.precision * 100).toFixed(1)}%</h3>
                    </div>
                  </Col>
                  <Col md={3} sm={6} className="mb-3">
                    <div className="metric-card text-center">
                      <h6 className="text-muted">Recall</h6>
                      <h3 className="text-info">{(metrics.recall * 100).toFixed(1)}%</h3>
                    </div>
                  </Col>
                  <Col md={3} sm={6} className="mb-3">
                    <div className="metric-card text-center">
                      <h6 className="text-muted">F1 Score</h6>
                      <h3 className="text-warning">{(metrics.f1_score * 100).toFixed(1)}%</h3>
                    </div>
                  </Col>
                </Row>
                <p className="text-muted text-center mt-2">Evaluated on {metrics.test_samples} test videos</p>
              </Card.Body>
            </Card>
          )}
        </Col>
      </Row>
    </Container>
  );
};

export default Upload;
